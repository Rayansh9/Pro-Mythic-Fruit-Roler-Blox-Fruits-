local PARENT
local parentUsed = "none"
do
	local ScreenGui = Instance.new("ScreenGui")
	ScreenGui.Name = "MythicAdminGui"
	ScreenGui.ResetOnSpawn = false
	ScreenGui.DisplayOrder = 999999999

	local function tryParent(target)
		if target then
			local ok = pcall(function()
				ScreenGui.Parent = target
			end)
			return ok and ScreenGui.Parent ~= nil
		end
		return false
	end

	local gethuiFunc = typeof(gethui) == "function" and gethui or (typeof(get_hidden_gui) == "function" and get_hidden_gui or nil)
	if gethuiFunc then
		local ok, hui = pcall(gethuiFunc)
		if ok and hui and tryParent(hui) then
			parentUsed = "gethui"
		end
	end
	if not ScreenGui.Parent and syn and syn.protect_gui then
		pcall(function()
			syn.protect_gui(ScreenGui)
			if CoreGuiService then ScreenGui.Parent = CoreGuiService end
		end)
		if ScreenGui.Parent then parentUsed = "syn.protect_gui" end
	end
	if not ScreenGui.Parent and CoreGuiService and tryParent(CoreGuiService) then
		parentUsed = "CoreGui"
	end
	if not ScreenGui.Parent and CoreGuiService then
		local robloxGui = CoreGuiService:FindFirstChild("RobloxGui")
		if robloxGui and tryParent(robloxGui) then parentUsed = "CoreGui.RobloxGui" end
	end
	if not ScreenGui.Parent and playerGui and tryParent(playerGui) then
		parentUsed = "PlayerGui"
	end
	if not ScreenGui.Parent then
		local sg = pcall(game.GetService, game, "StarterGui") and CoreGuiService or StarterGui
		if StarterGui and tryParent(StarterGui) then parentUsed = "StarterGui" end
	end
	PARENT = ScreenGui
	print("[MythicFruit] GUI parented to: " .. parentUsed)
end

	local function goTab(name)
		activeTab = name
		for _, tabInfo in ipairs(tabs) do
			local btn = tabButtons[tabInfo.Name]
			local isActive = tabInfo.Name == name
			if isActive then
				btn.BackgroundColor3 = Color3.fromRGB(80, 45, 120)
				local hasGrad = false
				for _, c in ipairs(btn:GetChildren()) do
					if c:IsA("UIGradient") then hasGrad = true end
				end
				if not hasGrad then
					addGradient(btn, Color3.fromRGB(110, 50, 170), Color3.fromRGB(60, 25, 90), 45)
				end
				btn:FindFirstChild("ActiveBar").Visible = true
				btn:FindFirstChild("Icon").BackgroundColor3 = THEME.purple
				btn:FindFirstChild("Icon"):FindFirstChild("TextLabel").TextColor3 = THEME.white
			else
				btn.BackgroundColor3 = Color3.fromRGB(40, 40, 46)
				for _, c in ipairs(btn:GetChildren()) do
					if c:IsA("UIGradient") then c:Destroy() end
				end
				btn:FindFirstChild("ActiveBar").Visible = false
				btn:FindFirstChild("Icon").BackgroundColor3 = THEME.gray
				btn:FindFirstChild("Icon"):FindFirstChild("TextLabel").TextColor3 = THEME.white
			end
		end
		for pname, page in pairs(pages) do
			page.Visible = (pname == name)
		end
		pcall(function()
			local page = pages[name]
			if page and page.AbsoluteSize.Y > 0 then
				contentScroller.CanvasSize = UDim2.new(0, 0, 0, page.AbsoluteSize.Y + 20)
			end
		end)
	end

	local function executeUrl(url)
		if not url or url == "" then
			loaderStatus.Text = "Status: No URL provided"
			loaderStatus.TextColor3 = THEME.red
			return
		end
		loaderStatus.Text = "Status: Fetching script..."
		loaderStatus.TextColor3 = THEME.accent
		local fetchSuccess, chunk = pcall(function()
			return game:HttpGet(url)
		end)
		if not fetchSuccess then
			loaderStatus.Text = "Status: Failed to fetch URL"
			loaderStatus.TextColor3 = THEME.red
			return
		end
		local loadSuccess, loaded = pcall(loadstring, chunk)
		if not loadSuccess then
			loaderStatus.Text = "Status: Compile error"
			loaderStatus.TextColor3 = THEME.red
			return
		end
		local runSuccess, runErr = pcall(loaded)
		if runSuccess then
			loaderStatus.Text = "Status: Executed successfully!"
			loaderStatus.TextColor3 = THEME.accent
			local seen = {}
			local clean = {}
			table.insert(urlHistory, 1, url)
			for _, u in ipairs(urlHistory) do
				if not seen[u] then
					seen[u] = true
					table.insert(clean, u)
				end
			end
			while #clean > 10 do table.remove(clean) end
			urlHistory = clean
			saveHistory()
			refreshHistory()
		else
			loaderStatus.Text = "Status: Runtime error\n" .. tostring(runErr)
			loaderStatus.TextColor3 = THEME.red
		end
	end

	local function giveFruitToInventory(fruitName)
		local found
		for _, container in ipairs({ReplicatedStorage, workspace, player.StarterGear}) do
			local asset = container:FindFirstChild(fruitName)
			if asset then
				found = asset
				break
			end
		end

		end

		return false
	end

	local function processCommand(text)
		local command = (text or ""):lower()

		if command:sub(1, 7) == ";speed " then
			local speedVal = tonumber(text:sub(8))
			if speedVal and player.Character and player.Character:FindFirstChild("Humanoid") then
				player.Character.Humanoid.WalkSpeed = speedVal
				applySuperSpeed(false)
				setOutput("Command Executed:\nWalkSpeed set to " .. tostring(speedVal), THEME.accent)
			else
				setOutput("Invalid speed or no character.", THEME.red)
			end

		elseif command == ";reset" then
			if player.Character and player.Character:FindFirstChild("Humanoid") then
				player.Character.Humanoid.Health = 0
				setOutput("Command Executed: Reset Character", THEME.accent)
			else
				setOutput("No character found.", THEME.red)
			end

		elseif command == ";roll" or command == ";spin" or command == ";gacha" then
			doGacha()

		elseif command == ";inv" or command == ";inventory" then
			goTab("Items")
			showInventory()

		elseif command == ";clearinv" then
			inventory = {}
			refreshInventory()
			setOutput("Inventory cleared.", THEME.accent)

		elseif command:sub(1, 7) == ";luck " then
			local v = tonumber(text:sub(8))
			if v and v > 0 then
				currentLuck = v
				updateStats()
				setOutput("Luck multiplier set to " .. tostring(v) .. "x", THEME.accent)
			else
				setOutput("Usage: ;luck <number>", THEME.red)
			end

		elseif command:sub(1, 8) == ";addbeli" then
			local v = tonumber(text:sub(9))
			if v then
				localBeli = localBeli + v
				updateStats()
				setOutput("Added " .. tostring(v) .. " Beli.\nBalance: " .. tostring(localBeli), THEME.accent)
			else
				setOutput("Usage: ;addbeli <number>", THEME.red)
			end

		elseif command == ";beli" then
			setOutput("FREE ROLLS ENABLED\nBeli balance: " .. tostring(localBeli) .. " (cosmetic only)", THEME.text)

		elseif command:sub(1, 6) == ";give " then
			local fruitName = text:sub(7)
			if fruitName and fruitName ~= "" then
				giveFruitToInventory(fruitName)
				if not table.find(inventory, fruitName) then table.insert(inventory, fruitName) end
				refreshInventory()
				setOutput("Gave " .. fruitName .. "\nto your inventory (FREE)!", THEME.accent)
			else
				setOutput("Usage: ;give <FruitName>", THEME.red)
			end

		elseif command == ";help" then
			setOutput(";speed <n>  |  ;reset\n;roll (FREE)  |  ;inv  |  ;clearinv\n;luck <n>  |  ;addbeli <n>\n;give <fruit>  |  ;beli\n;help", THEME.text)

		else
			setOutput("Unknown Command: " .. tostring(text), THEME.red)
		end
	end

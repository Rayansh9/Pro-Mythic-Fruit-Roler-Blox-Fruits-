local PARENT
local parentUsed = "none"
do
	local ScreenGui = Instance.new("ScreenGui")
	ScreenGui.Name = "MythicAdminGui"
	ScreenGui.ResetOnSpawn = false
	ScreenGui.DisplayOrder = 999999999

	local function tryParent(target)
		if target then
			local ok = pcall(function()
				ScreenGui.Parent = target
			end)
			return ok and ScreenGui.Parent ~= nil
		end
		return false
	end

	local gethuiFunc = typeof(gethui) == "function" and gethui or (typeof(get_hidden_gui) == "function" and get_hidden_gui or nil)
	if gethuiFunc then
		local ok, hui = pcall(gethuiFunc)
		if ok and hui and tryParent(hui) then
			parentUsed = "gethui"
		end
	end
	if not ScreenGui.Parent and syn and syn.protect_gui then
		pcall(function()
			syn.protect_gui(ScreenGui)
			if CoreGuiService then ScreenGui.Parent = CoreGuiService end
		end)
		if ScreenGui.Parent then parentUsed = "syn.protect_gui" end
	end
	if not ScreenGui.Parent and CoreGuiService and tryParent(CoreGuiService) then
		parentUsed = "CoreGui"
	end
	if not ScreenGui.Parent and CoreGuiService then
		local robloxGui = CoreGuiService:FindFirstChild("RobloxGui")
		if robloxGui and tryParent(robloxGui) then parentUsed = "CoreGui.RobloxGui" end
	end
	if not ScreenGui.Parent and playerGui and tryParent(playerGui) then
		parentUsed = "PlayerGui"
	end
	if not ScreenGui.Parent then
		local sg = pcall(game.GetService, game, "StarterGui") and CoreGuiService or StarterGui
		if StarterGui and tryParent(StarterGui) then parentUsed = "StarterGui" end
	end
	PARENT = ScreenGui
	print("[MythicFruit] GUI parented to: " .. parentUsed)
end

	local function executeUrl(url)
		if not url or url == "" then
			loaderStatus.Text = "Status: No URL provided"
			loaderStatus.TextColor3 = THEME.red
			return
		end
		loaderStatus.Text = "Status: Fetching script..."
		loaderStatus.TextColor3 = THEME.accent
		local fetchSuccess, chunk = pcall(function()
			return game:HttpGet(url)
		end)
		if not fetchSuccess then
			loaderStatus.Text = "Status: Failed to fetch URL"
			loaderStatus.TextColor3 = THEME.red
			return
		end
		local loadSuccess, loaded = pcall(loadstring, chunk)
		if not loadSuccess then
			loaderStatus.Text = "Status: Compile error"
			loaderStatus.TextColor3 = THEME.red
			return
		end
		local runSuccess, runErr = pcall(loaded)
		if runSuccess then
			loaderStatus.Text = "Status: Executed successfully!"
			loaderStatus.TextColor3 = THEME.accent
			local seen = {}
			local clean = {}
			table.insert(urlHistory, 1, url)
			for _, u in ipairs(urlHistory) do
				if not seen[u] then
					seen[u] = true
					table.insert(clean, u)
				end
			end
			while #clean > 10 do table.remove(clean) end
			urlHistory = clean
			saveHistory()
			refreshHistory()
		else
			loaderStatus.Text = "Status: Runtime error\n" .. tostring(runErr)
			loaderStatus.TextColor3 = THEME.red
		end
	end

	end)

	local backpack = player:WaitForChild("Backpack", 15) or Instance.new("Backpack")

	local function createFruitTool(fruitName)
		local tool = Instance.new("Tool")
		tool.Name = fruitName
		tool.RequiresHandle = false
		tool.CanBeDropped = true
		local handle = Instance.new("Part")
		handle.Name = "Handle"
		handle.Size = Vector3.new(1.2, 1.2, 1.2)
		handle.Material = Enum.Material.Neon
		handle.Color = Color3.fromRGB(140, 50, 230)
		handle.CanCollide = false
		handle.CanQuery = false
		handle.CanTouch = false
		handle.Parent = tool
		local fruitLabel = Instance.new("BillboardGui")
		fruitLabel.Size = UDim2.new(0, 6, 0, 1.5)
		fruitLabel.StudsOffset = Vector3.new(0, 1.5, 0)
		fruitLabel.AlwaysOnTop = true
		local label = Instance.new("TextLabel")
		label.Size = UDim2.new(1, 0, 1, 0)
		label.BackgroundTransparency = 1
		label.Text = fruitName
		label.TextColor3 = THEME.accent
		label.TextStrokeTransparency = 0
		label.TextSize = 14
		label.Font = Enum.Font.GothamBold
		label.Parent = fruitLabel
		fruitLabel.Parent = handle
		tool.Parent = backpack
		return tool
	end

	local function giveFruitToInventory(fruitName)
		local found
		for _, container in ipairs({ReplicatedStorage, workspace, player.StarterGear}) do
			local asset = container:FindFirstChild(fruitName)
			if asset then
				found = asset
				break
			end
		end
		if found then
			local clone = found:Clone()
			clone.Parent = backpack
			return true
		end
		createFruitTool(fruitName)
		return false
	end

	local function doGacha()
		if localBeli < ROLL_COST then
			setOutput("Insufficient Beli!\nCost: 100,000 | Balance: " .. tostring(localBeli), THEME.red)
			return
		end
		localBeli = localBeli - ROLL_COST
		local result = rollFruit()
		table.insert(inventory, result.Name)
		giveFruitToInventory(result.Name)
		beliLabel.Text = "Beli: " .. tostring(localBeli) .. "  |  Luck: " .. tostring(currentLuck) .. "x"
		setOutput("SPIN RESULT: " .. result.Name .. "\nRarity: " .. result.Rarity .. "\nGIVEN TO YOUR INVENTORY!", THEME.accent)
		refreshInventory()
	end

	local function showInventory()
		if #inventory == 0 then
			setOutput("Inventory is empty.\nUse ;roll to spin!", THEME.text)
		else
			local lines = {}
			for i, name in ipairs(inventory) do
				table.insert(lines, i .. ". " .. name)
			end
			setOutput("INVENTORY (" .. #inventory .. ")\n" .. table.concat(lines, "\n"), THEME.text)
		end
	end

	cmdBox.FocusLost:Connect(function(enterPressed)
		if not enterPressed then return end
		local text = cmdBox.Text
		cmdBox.Text = ""
		local command = text:lower()

		if command:sub(1, 7) == ";speed " then
			local speedVal = tonumber(text:sub(8))
			if speedVal and player.Character and player.Character:FindFirstChild("Humanoid") then
				player.Character.Humanoid.WalkSpeed = speedVal
				setOutput("Command Executed:\nWalkSpeed set to " .. tostring(speedVal), THEME.accent)
			else
				setOutput("Invalid speed or no character.", THEME.red)
			end

		elseif command == ";reset" then
			if player.Character and player.Character:FindFirstChild("Humanoid") then
				player.Character.Humanoid.Health = 0
				setOutput("Command Executed: Reset Character", THEME.accent)
			else
				setOutput("No character found.", THEME.red)
			end

		elseif command == ";roll" or command == ";spin" or command == ";gacha" then
			doGacha()

		elseif command == ";inv" or command == ";inventory" then
			invFrame.Visible = true
			showInventory()

		elseif command == ";clearinv" then
			inventory = {}
			refreshInventory()
			setOutput("Inventory cleared.", THEME.accent)

		elseif command:sub(1, 7) == ";luck " then
			local v = tonumber(text:sub(8))
			if v and v > 0 then
				currentLuck = v
				beliLabel.Text = "Beli: " .. tostring(localBeli) .. "  |  Luck: " .. tostring(currentLuck) .. "x"
				setOutput("Luck multiplier set to " .. tostring(v) .. "x", THEME.accent)
			else
				setOutput("Usage: ;luck <number>", THEME.red)
			end

		elseif command:sub(1, 8) == ";addbeli" then
			local v = tonumber(text:sub(9))
			if v then
				localBeli = localBeli + v
				beliLabel.Text = "Beli: " .. tostring(localBeli) .. "  |  Luck: " .. tostring(currentLuck) .. "x"
				setOutput("Added " .. tostring(v) .. " Beli.\nBalance: " .. tostring(localBeli), THEME.accent)
			else
				setOutput("Usage: ;addbeli <number>", THEME.red)
			end

		elseif command == ";beli" then
			setOutput("Balance: " .. tostring(localBeli) .. " Beli", THEME.text)

		elseif command:sub(1, 6) == ";give " then
			local fruitName = text:sub(7)
			if fruitName and fruitName ~= "" then
				giveFruitToInventory(fruitName)
				table.insert(inventory, fruitName)
				refreshInventory()
				setOutput("Gave " .. fruitName .. "\nto your inventory!", THEME.accent)
			else
				setOutput("Usage: ;give <FruitName>", THEME.red)
			end

		elseif command == ";help" then
			setOutput(";speed <n>  |  ;reset\n;roll  |  ;inv  |  ;clearinv\n;luck <n>  |  ;addbeli <n>\n;give <fruit>  |  ;beli\n;help", THEME.text)

		else
			setOutput("Unknown Command: " .. text, THEME.red)
		end
	end)

	-- ============================================================
	-- Section 7: Chat Hooks (!spin / !roll / !gacha)
	-- ============================================================

	local function onChatTrigger()
		doGacha()
	end

	pcall(function()
		if TextChatService and TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
			local channels = TextChatService:FindFirstChild("TextChannels")
			if channels then
				local general = channels:FindFirstChild("RBXGeneral")
				if general then
					general.OnSendMessage:Connect(function(message)
						local lower = message.Text:lower()
						if lower == "!spin" or lower == "!roll" or lower == "!gacha" then
							onChatTrigger()
						end
					end)
				end
			end
		end
	end)

	pcall(function()
		player.Chatted:Connect(function(message)
			local lower = message:lower()
			if lower == "!spin" or lower == "!roll" or lower == "!gacha" then
				onChatTrigger()
			end
		end)
	end)

	setOutput("PRO Admin System Active.\nCreated by: " .. player.Name, THEME.accent)

	-- ============================================================
	-- Section 8: Show/hide hotkey (Right Alt)
	-- ============================================================

	pcall(function()
		UserInputService.InputBegan:Connect(function(input, gameProcessed)
			if gameProcessed then return end
			if input.KeyCode == Enum.KeyCode.RightAlt then
				PARENT.Enabled = not PARENT.Enabled
			end
		end)
	end)
